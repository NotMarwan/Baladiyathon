**Goal:**

(function (root, factory) {
  if (typeof module === 'object' && module.exports) {
    module.exports = factory();
  } else {
    root.MasarDecision = factory();
  }
})(typeof self !== 'undefined' ? self : this, function () {
  'use strict';

  const REQUIRED_FIELDS = [
    'permitId',
    'segmentId',
    'startDate',
    'startHour',
    'durationHours',
    'lanes',
    'lanesClosed',
    'aadt',
    'originNodeId',
    'destinationNodeId'
  ];

  function validateDecisionInput(input) {
    const value = input && typeof input === 'object' ? input : {};
    const missing = REQUIRED_FIELDS.filter((key) =>
      value[key] === undefined || value[key] === null || value[key] === ''
    );
    return {
      status: missing.length ? 'blocked' : 'ready',
      missing,
      assumptions: Array.isArray(value.assumptions) ? value.assumptions : [],
      canDecide: missing.length === 0
    };
  }

  function createDecisionService(dependencies) {
    const deps = dependencies || {};
    const required = ['engine', 'routing', 'forecast', 'reasons', 'conflict'];
    required.forEach((name) => {
      if (!deps[name]) throw new TypeError(`Missing dependency: ${name}`);
    });

    return {
      evaluate(input) {
        const qualityGate = validateDecisionInput(input);
        if (!qualityGate.canDecide) {
          return {
            qualityGate,
            baseline: null,
            scheduleAlternatives: [],
            routeAlternatives: [],
            forecast: null,
            conflicts: [],
            verdict: null,
            evidence: []
          };
        }

        const baseline = deps.engine.score(input);
        const scheduleAlternatives = deps.engine.optimize(input);
        const routeAlternatives = deps.routing.alternatives(input);
        const predicted = deps.forecast.predict(input);
        const conflicts = deps.conflict.detect(input);
        const explanations = deps.reasons.explain({
          input,
          baseline,
          scheduleAlternatives,
          routeAlternatives,
          forecast: predicted,
          conflicts
        });
        const evidence = explanations.map((item) => ({
          label: item.label,
          value: Number(item.contribution),
          provenance: item.provenance || 'local-computation'
        }));

        return {
          qualityGate,
          baseline,
          scheduleAlternatives,
          routeAlternatives,
          forecast: predicted,
          conflicts,
          verdict: {
            status: conflicts.length ? 'conditional' : 'recommended',
            scheduleId: scheduleAlternatives[0] && scheduleAlternatives[0].id,
            routeId: routeAlternatives[0] && routeAlternatives[0].id
          },
          evidence
        };
      }
    };
  }

  return { createDecisionService, validateDecisionInput };
});
